Cập nhật dữ liệu vào bộ nhớ cục bộ if (prop === 'netType' || prop === 'racType') { room[prop] = val; } else { const parsedVal = parseNumber(val); if (prop === 'giaPhong' && Number(room.giaPhong) !== Number(parsedVal)) { room.giaPhong = parsedVal; room.duCoc = false; } else { room[prop] = parsedVal; } } // 2. Map tên thuộc tính JS sang tên cột thực tế trên Supabase const dbFieldMap = { giaPhong: 'gia_phong', netType: 'net_type', racType: 'rac_type', dienCu: 'dien_cu', dienMoi: 'dien_moi', nuocCu: 'nuoc_cu', nuocMoi: 'nuoc_moi', tienKhac: 'tien_khac' }; const dbColumn = dbFieldMap[prop]; const updateValue = (prop === 'netType' || prop === 'racType') ? val : parseNumber(val); // 3. Đẩy trực tiếp 1 dòng bị thay đổi lên Supabase ngay lập tức if (dbColumn) { try { await supabaseClient .from('phong_trans_tro') .update({ [dbColumn]: updateValue }) .eq('id', id); } catch (err) { console.error("Lỗi lưu dữ liệu lên Supabase:", err); } } // 4. Cập nhật các con số tính toán thống kê trên giao diện updateGeneralStatsCards(); }
